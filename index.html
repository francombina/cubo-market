Yahoo Finance API Handler (with CORS proxy handling) async function fetchYahooQuotes() { const tbody = document.getElementById("stocks-tbody"); stocksDataStore = []; for (const item of WATCHLIST_STOCKS) { try { // Public CORS Proxy wrapper for client-side Yahoo requests const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(`https://query1.finance.yahoo.com/v8/finance/chart/${item.ticker}`)}`; const res = await fetch(proxyUrl); if (res.ok) { const wrapper = await res.json(); const json = JSON.parse(wrapper.contents); const meta = json.chart?.result?.[0]?.meta; if (meta) { const price = meta.regularMarketPrice || 0; const prevClose = meta.chartPreviousClose || meta.previousClose || price; const changePct = prevClose ? ((price - prevClose) / prevClose) * 100 : 0; const volume = meta.regularMarketVolume || 0; stocksDataStore.push({ ticker: item.ticker, name: item.name, price: price, changePct: changePct, volume: volume, source: item.source }); } } } catch (e) { console.warn(`Yahoo fetch error for ${item.ticker}:`, e); } } renderStocksTable(stocksDataStore); } // Render Stocks Table function renderStocksTable(data) { const tbody = document.getElementById("stocks-tbody"); if (!data || data.length === 0) { tbody.innerHTML = `No se pudieron cargar cotizaciones en vivo en este momento.`; return; } tbody.innerHTML = data.map(stock => { const isPositive = stock.changePct >= 0; const changeClass = isPositive ? "text-emerald-400" : "text-rose-400"; const changeSign = isPositive ? "+" : ""; return ` ${stock.ticker} ${stock.name} $${stock.price.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} ${changeSign}${stock.changePct.toFixed(2)}% ${stock.volume ? stock.volume.toLocaleString() : 'N/A'} ${stock.source} `; }).join(''); } // Filter Table Search function filterStocks() { const query = document.getElementById("search-stocks").value.toLowerCase(); const filtered = stocksDataStore.filter(s => s.ticker.toLowerCase().includes(query) || s.name.toLowerCase().includes(query) ); renderStocksTable(filtered); } // 3. SEC EDGAR Official API Handler async function fetchSECFundamentals() { const secTbody = document.getElementById("sec-tbody"); let rowsHtml = ""; for (const company of SEC_COMPANIES) { try { const secUrl = `https://data.sec.gov/api/xbrl/companyfacts/CIK${company.CIK}.json`; const response = await fetch(secUrl, { headers: SEC_HEADERS }); if (response.ok) { const data = await response.json(); const usGaap = data.facts?.['us-gaap']; // Extract Revenues / Operating Income metrics dynamically let latestRevenues = "N/D"; if (usGaap?.Revenues?.units?.USD) { const revUnits = usGaap.Revenues.units.USD; const lastRev = revUnits[revUnits.length - 1]; latestRevenues = `$${(lastRev.val / 1e9).toFixed(2)} B (${lastRev.fy} Q${lastRev.fp || ''})`; } rowsHtml += ` ${company.name} (${company.ticker}) ${company.CIK} ${latestRevenues} Oficial XBRL SEC OK 200 `; } else { throw new Error(`SEC HTTP ${response.status}`); } } catch (err) { rowsHtml += ` ${company.name} (${company.ticker}) ${company.CIK} Consulta directa de SEC disponible en la red Petición Registrada `; } } secTbody.innerHTML = rowsHtml; } // Status Indicator Helper function updateSyncStatus(type, message) { const indicator = document.getElementById("sync-indicator"); const text = document.getElementById("sync-text"); text.innerText = message; if (type === "loading") { indicator.className = "w-2 h-2 rounded-full bg-amber-500 animate-pulse"; } else if (type === "online") { indicator.className = "w-2 h-2 rounded-full bg-emerald-500"; } else { indicator.className = "w-2 h-2 rounded-full bg-rose-500"; } }
